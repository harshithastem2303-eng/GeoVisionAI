# GeoVision backend.
#
#   python -m venv .venv
#   # macOS / Linux:   source .venv/bin/activate
#   # Windows:         .venv\Scripts\activate
#   pip install -r backend/requirements.txt
#
# Python 3.10+.

# --- API ------------------------------------------------------------------
fastapi
uvicorn[standard]
pydantic>=2
python-multipart

# --- Vision ---------------------------------------------------------------
opencv-python
numpy
ultralytics          # YOLO + BoT-SORT

# --- Camera ---------------------------------------------------------------
# Optional. The backend imports it lazily and runs with
# GEOVISION_CAMERA_BACKEND=mock when it is absent or no camera is attached.
# Not available as a wheel on every platform/Python combination.
pyrealsense2 ; platform_machine != "arm64" or sys_platform != "darwin"

# --- Database -------------------------------------------------------------
psycopg2-binary

# --- WASTRAQ integration --------------------------------------------------
# None. The outbound client uses urllib from the standard library on
# purpose: one JSON POST does not justify another wheel that has to install
# on both a Windows laptop and a Mac.

# --- Tests ----------------------------------------------------------------
# pytest is only needed to run the suite:
#     pip install pytest && pytest backend/tests -v
# The tests are written as unittest.TestCase, so they also run with
#     python -m unittest discover -s backend/tests -t backend
pytest
